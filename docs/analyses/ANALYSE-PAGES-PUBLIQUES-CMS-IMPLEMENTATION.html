Objectif A — Enrichissement entité

Ajouter les champs manquants à Etablissement : localisation géographique (pays, région, quartier, GPS), identifiants officiels (enregistrement, identification, autorisation), et informations complémentaires sans redondance.

Objectif B — CMS White-Label

Permettre à chaque établissement de personnaliser intégralement sa page publique : contenu, sections, médias, thème visuel, navigation, styles. Système CMS complet intégré à la plateforme.

Objectif C — Routing path-based

Pages accessibles via /e/:code (codeEtablissement comme slug). Zero DNS, CDN-compatible, SEO-friendly. Middleware d'authentification optionnel pour enrichir le contexte.

ℹ
Décision architecturale validée : Le routing path-based (/e/:code) est choisi pour sa simplicité opérationnelle. Le champ codeEtablissement (varchar(50), unique) sert de slug URL naturel.

    alt Cache hit
        C-->>A: Page JSON cached
    else Cache miss

Système CMS — Architecture

4.1 — Principes de conception

Isolation multi-tenant stricte

Chaque entité CMS porte un etablissementId. RLS PostgreSQL + middleware tenant. Aucun établissement ne peut voir ou modifier le contenu d'un autre.

Personnalisation extrême

Sections réordonnables par drag & drop, thèmes visuels complets (couleurs, polices, espacements), widgets configurables, pages illimitées avec sous-pages.

Versioning & historique

Chaque modification crée un snapshot dans cms_versions. Possibilité de restaurer n'importe quelle version antérieure. Brouillon → Publication.

Performance par cache

Pages publiques servies depuis Redis (TTL 300s). Invalidation automatique à chaque modification CMS. Cache-Control HTTP pour le navigateur.

Entités CMS — Schéma complet

5.1 — CmsPage (pages de l'établissement)

@Entity('cms_pages')
@Index(['etablissementId', 'slug'], { unique: true })
export class CmsPage {
    @PrimaryGeneratedColumn('uuid')
    id!: string;

    @Column({ type: 'uuid' })
    etablissementId!: string;

    @Column({ type: 'varchar', length: 200 })
    titre!: string;

    @Column({ type: 'varchar', length: 200 })
    slug!: string;  // ex: 'a-propos', 'contact', 'inscriptions'

    @Column({ type: 'varchar', length: 50, default: 'custom' })
    template!: string;  // accueil | custom | galerie | actualites | contact | inscriptions

    @Column({ type: 'varchar', length: 20, default: 'BROUILLON' })
    statut!: 'BROUILLON' | 'PUBLIE' | 'ARCHIVE';

    @Column({ type: 'int', default: 0 })
    ordre!: number;

    @Column({ type: 'uuid', nullable: true })
    pageParentId?: string;  // sous-pages

    @Column({ type: 'jsonb', nullable: true })
    metadata?: Record<string, unknown>;  // données custom

    @Column({ type: 'jsonb', nullable: true })
    seo?: {  // SEO
        metaTitle?: string;
        metaDescription?: string;
        metaKeywords?: string[];
        ogImage?: string;
    };

    @Column({ type: 'boolean', default: false })
    estPageAccueil!: boolean;  // une seule par établissement

    @ManyToOne(() => CmsPage, { nullable: true })
    @JoinColumn({ name: 'pageParentId' })
    pageParent?: CmsPage;

    @OneToMany(() => CmsSection, 'page', { cascade: true })
    sections!: CmsSection[];

    @CreateDateColumn() createdAt!: Date;
    @UpdateDateColumn() updatedAt!: Date;
}

5.2 — CmsSection (blocs de contenu)

@Entity('cms_sections')
export class CmsSection {
    @PrimaryGeneratedColumn('uuid')
    id!: string;

    @Column({ type: 'uuid' })
    pageId!: string;

    @Column({ type: 'varchar', length: 50 })
    type!: SectionType;  // enum ci-dessous

    @Column({ type: 'jsonb' })
    contenu!: Record<string, unknown>;  // contenu structuré par type

    @Column({ type: 'int', default: 0 })
    ordre!: number;

    @Column({ type: 'jsonb', nullable: true })
    styles?: Record<string, unknown>;  // styles custom (padding, bg, etc.)

    @Column({ type: 'boolean', default: true })
    visible!: boolean;

    @Column({ type: 'varchar', length: 50, nullable: true })
    anchorId?: string;  // pour navigation intra-page

    @ManyToOne(() => CmsPage, 'sections')
    @JoinColumn({ name: 'pageId' })
    page!: CmsPage;

    @CreateDateColumn() createdAt!: Date;
    @UpdateDateColumn() updatedAt!: Date;
}

Thèmes & Design System

6.1 — Templates de pages prédéfinis

🏠 Accueil (accueil)

Hero plein écran + chiffres clés + présentation + actualités récentes + CTA inscription + partenaires. Layout: Hero → Sections → Footer.

📄 Page standard (custom)

Header page + sections libres + sidebar widgets. Layout: Breadcrumb → Contenu → Sidebar.

🖼️ Galerie (galerie)

Filtres par album + grille responsive + lightbox. Layout: Filtres → Grille masonry → Lightbox.

📰 Actualités (actualites)

Liste paginée + filtres par catégorie + pagination infinite scroll.

📞 Contact (contact)

Formulaire + carte GPS + infos contact + horaires. Layout: 2 colonnes (form + map).

📝 Inscriptions (inscriptions)

Processus multi-étapes + documents requis + FAQ + CTA. Layout: Stepper → Détails → FAQ.

async getPagePublique(code: string): Promise<PagePubliqueDto> {
    const etab = await this.etabRepo.createQueryBuilder('e')
        .select([
            'e.id', 'e.nom', 'e.slogan', 'e.pays', 'e.region',
            'e.ville', 'e.quartier', 'e.adresse',
            'e.latitude', 'e.longitude',
            'e.contactEmail', 'e.contactTelephone',
            'e.logoBase64', 'e.logoType',
            'e.couleurPrimaire', 'e.couleurSecondaire',
            'e.siteWeb', 'e.facebook', 'e.twitter',
            'e.heuresOuverture', 'e.heuresFermeture',
            'e.descriptionPublique', 'e.codeEtablissement',
            'e.sousSysteme', 'e.type',
        ])
        .where('e.codeEtablissement = :code', { code })
        .andWhere('e.actif = true')
        .getOne();

    if (!etab) throw new AppError('Établissement introuvable', 404);
    return etab;
    // NOTE: numeroContribuable, numeroCompteBancaire, numeroArrete,
    //       numeroEnregistrement, numeroAutorisation EXCLUS (données privées)
}

13.3 — Migration permissions CMS

-- backend/database/migrations/182-cms-permissions.sql
-- Permissions RBAC pour le module CMS

-- Création des permissions
DO $$
DECLARE
    perms text[] := ARRAY[
        'cms:pages:view', 'cms:pages:create', 'cms:pages:edit',
        'cms:pages:publish', 'cms:pages:delete',
        'cms:sections:view', 'cms:sections:create', 'cms:sections:edit',
        'cms:sections:delete',
        'cms:medias:view', 'cms:medias:upload', 'cms:medias:delete',
        'cms:themes:view', 'cms:themes:manage',
        'cms:menus:manage',
        'cms:widgets:manage',
        'cms:versions:view', 'cms:versions:restore'
    ];
    p text;
    mod_id uuid;
BEGIN
    -- Module CMS dans le catalogue
    INSERT INTO modules_catalogue (code, nom, categorie, actifParDefaut, estSysteme)
    VALUES ('cms', 'CMS — Pages publiques', 'PREMIUM', false, true)
    ON CONFLICT (code) DO NOTHING;

    SELECT id INTO mod_id FROM modules_catalogue WHERE code = 'cms';

    FOREACH p IN ARRAY perms
    LOOP
        INSERT INTO permissions (code, libelle, module, "moduleCatalogueId")
        VALUES (p, initcap(split_part(p, ':', 2)) || ' — ' || split_part(p, ':', 3),
                'cms', mod_id)
        ON CONFLICT (code) DO NOTHING;
    END LOOP;

    -- Attribution ADMIN + CHEF_ETABLISSEMENT (toutes permissions)
    INSERT INTO role_permissions ("roleId", "permissionId")
    SELECT r.id, p.id
    FROM roles r
    CROSS JOIN permissions p
    WHERE r.code IN ('ADMIN', 'CHEF_ETABLISSEMENT')
      AND p.module = 'cms'
    ON CONFLICT DO NOTHING;

    -- Attribution DIRECTEUR (sauf publish, themes:manage, widgets:manage, versions:restore)
    INSERT INTO role_permissions ("roleId", "permissionId")
    SELECT r.id, p.id
    FROM roles r
    CROSS JOIN permissions p
    WHERE r.code = 'DIRECTEUR'
      AND p.module = 'cms'
      AND p.code NOT IN ('cms:themes:manage', 'cms:widgets:manage', 'cms:versions:restore')
    ON CONFLICT DO NOTHING;
END $$;
